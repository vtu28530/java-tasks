import java.util.Arrays;

public class Task2 {

    public static void main(String[] args) {

        int[] arr = {7, 10, 4, 3, 20, 15};

        int k = 3;

        Arrays.sort(arr);

        int kthSmallest = arr[k - 1];

        System.out.println("The " + k + "th smallest element is: " + kthSmallest);
    }
}
